filter

filterはtrueになった要素のみを返すものらしい．使いこなせたらかなり便利そう．

{% highlight js %} let a = []; for (let i = 0; i < 10; ++i) { a.push(i); } const b = a.filter(x => { return x & 1; }); console.log(b); {% endhighlight %} {% highlight sh %} # 出力結果 > [ 1, 3, 5, 7, 9 ] {% endhighlight %}

map

mapは各要素に対してコールバック函数なるものを実行するらしい．

{% highlight js %} let a = []; let even = []; let odd = []; for (let i = 0; i < 10; ++i) { a.push(i); } a.map(x => { if (x & 1) { odd.push(x); } else { even.push(x); } }); console.log(odd); console.log(even); {% endhighlight %}
{% highlight sh %} # 出力結果 > [ 1, 3, 5, 7, 9 ] [ 0, 2, 4, 6, 8 ] {% endhighlight %}

returnして，各要素に何らかの操作をした配列を返すこともできる．

{% highlight js %} let a = []; for (let i = 0; i < 10; ++i) { a.push(i); } const b = a.map(x => { return Math.sqrt(x) }); console.log(b); {% endhighlight %}
{% highlight sh %} # 出力結果 > [ 0, 1, 1.4142135623730951, 1.7320508075688772, 2, 2.23606797749979, 2.449489742783178, 2.6457513110645907, 2.8284271247461903, 3 ] {% endhighlight %}

reduce

reduceもコールバック函数を呼び出すがmapとは少し違う．reduceに入るコールバック函数は，第一引数が直前のコールバック函数の戻り値，または初期値で，第二引数が現在の要素である． 従って，配列の総和とか総積とか，なんかそう言ったことに向いてそうである．

{% highlight js %} let a = []; for (let i = 0; i < 10; ++i) { a.push(i + 1); } const b = a.map(x => { return Math.log(x); }); const val = b.reduce((acc, cur) => { return acc + cur; }, 0); console.log(val); console.log(Math.pow(Math.E, val)) {% endhighlight %}
{% highlight sh %} # 出力結果 > 15.104412573075518 3628800.0000000056 {% endhighlight %}
